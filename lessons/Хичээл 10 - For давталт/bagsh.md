## Түгээмэл алдаа

| Алдаа | Засвар |
| ----- | ------ |
| Хуудас гацна | Хязгааргүй давталт: `i--`, `i = 1` (`==` биш), `while` дотор `sar + 1` мартсан. Таб хаа → засаад нээ |
| `let s = 0`-ийг давталтын **дотор** бичсэн | Давталт бүрт 0 болж байна — гадна гарга |
| `faktorial` → 0 | Үржвэр 1-ээс эхэлнэ |
| `niilber(5)` → 10 | `i < n` — `<=` хэрэгтэй |
| `return` давталт дотор | Эхний удаад л буцна — давталтын **дараа** |
| `urjver`: Хүлээж буй ≠ Чиний (харагдахдаа адилхан) | `×` тэмдэг (`x` биш), `=`-ийн хоёр талын зай |
| `toolokh`: `"5 4 3 2 1  Хөөрлөө!"` | Илүү зай — `s + i + " "` дараа нь `+ "Хөөрлөө!"` |

## Хариу — js-10.html

```js
function niilber(n) { let s = 0; for (let i = 1; i <= n; i++) { s = s + i; } return s; }
function odod(n) { let t = ""; for (let i = 1; i <= n; i++) { t = t + "⭐"; } return t; }
function toolokh(n) { let t = ""; for (let i = n; i >= 1; i--) { t = t + i + " "; } return t + "Хөөрлөө!"; }
function tegshToo(n) { let too = 0; for (let i = 1; i <= n; i++) { if (i % 2 === 0) { too = too + 1; } } return too; }
function urjver(n) {
  let t = "";
  for (let i = 1; i <= 10; i++) { t = t + n + " × " + i + " = " + n * i + "\n"; }
  return t;
}
function faktorial(n) { let u = 1; for (let i = 1; i <= n; i++) { u = u * i; } return u; }
function hurimtlal(zorilgo, sarBur) {
  let mungu = 0;
  let sar = 0;
  while (mungu < zorilgo) { mungu = mungu + sarBur; sar = sar + 1; }
  return sar;
}
function anhnii(n) {
  if (n < 2) { return "Анхны биш"; }
  for (let i = 2; i < n; i++) { if (n % i === 0) { return "Анхны биш"; } }
  return "Анхны тоо";
}
function tsifr(n) {
  let s = 0;
  while (n > 0) { s = s + n % 10; n = Math.floor(n / 10); }
  return s;
}
```

`anhnii`-д давталт дотор `return` нь **зөв**: хуваагч олдвол шууд дуусна. Ахисан хүүхдүүдтэй "яагаад энд болж байгаа юм бэ?" гэж ярилц.
